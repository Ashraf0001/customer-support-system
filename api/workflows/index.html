    class CustomerSupportWorkflow(Workflow):
    """Customer support ticket processing workflow.

    This workflow processes incoming customer support tickets through
    a series of AI-powered analysis and response generation steps:

    1. AnalyzeTicketNode: Analyzes ticket content for classification and sentiment
    2. ClassifyTicketNode: Validates and enhances classification results
    3. GenerateResponseNode: Generates appropriate response based on classification

    The workflow follows the lightweight MVP pattern with 3 nodes maximum
    and simple, linear processing flow.
    """

    workflow_schema = WorkflowSchema(
        description="Customer support ticket processing workflow for lightweight MVP",
        event_schema=SupportTicketSchema,
        start=AnalyzeTicketNode,
        nodes=[
            NodeConfig(
                node=AnalyzeTicketNode,
                connections=[ClassifyTicketNode],
                description="Analyzes incoming support tickets for classification and sentiment using Ollama"
            ),
            NodeConfig(
                node=ClassifyTicketNode,
                connections=[GenerateResponseNode],
                description="Validates and enhances classification results with business logic"
            ),
            NodeConfig(
                node=GenerateResponseNode,
                connections=[],
                description="Generates helpful responses based on classification using Ollama"
            ),
        ],
    )

    class AnalyzeTicketNode(Node):
    """Analyzes incoming support tickets for classification and sentiment.

    This node processes customer support tickets to extract key information
    such as intent classification, sentiment analysis, and priority scoring.
    It uses Ollama with Llama 3.1 8B for lightweight, local AI processing.
    """

    def __init__(self):
        self.ollama_url = f"{settings.ollama_base_url}/api/generate"
        self.model = settings.ollama_model
        self.start_time = None

    def process(self, task_context: TaskContext) -> TaskContext:
        """Process the ticket analysis workflow.

        Args:
            task_context: The task context containing the ticket data

        Returns:
            Updated TaskContext with analysis results

        Raises:
            OllamaConnectionError: If unable to connect to Ollama service
            InvalidTicketDataError: If ticket data is malformed
        """
        self.start_time = time.time()
        ticket = task_context.event

        logger.info(f"Starting ticket analysis", extra={
            "ticket_id": ticket.id,
            "node": self.node_name,
            "status": "started"
        })

        try:
            # Build the prompt using the template
            prompt = self._build_analysis_prompt(ticket)

            # Call Ollama API
            classification_result = self._call_ollama(prompt)

            # Parse the JSON response
            classification_data = self._parse_ollama_response(classification_result)

            # Calculate processing time
            processing_time = time.time() - self.start_time

            # Store results in task context
            task_context.nodes[self.node_name] = {
                "status": "completed",
                "classification": classification_data,
                "processing_time": processing_time,
                "model_used": self.model
            }

            logger.info(f"Ticket analysis completed", extra={
                "ticket_id": ticket.id,
                "node": self.node_name,
                "status": "completed",
                "processing_time": processing_time,
                "category": classification_data.get("category"),
                "sentiment": classification_data.get("sentiment")
            })

        except Exception as e:
            processing_time = time.time() - self.start_time if self.start_time else 0

            logger.error(f"Ticket analysis failed", extra={
                "ticket_id": ticket.id,
                "node": self.node_name,
                "status": "error",
                "error": str(e),
                "processing_time": processing_time
            })

            task_context.nodes[self.node_name] = {
                "status": "error",
                "error": str(e),
                "processing_time": processing_time
            }

        return task_context

    def _build_analysis_prompt(self, ticket) -> str:
        """Build the analysis prompt using the Jinja2 template.

        Args:
            ticket: The support ticket data

        Returns:
            Formatted prompt string
        """
        try:
            return PromptManager.get_prompt(
                "ticket_analysis",
                ticket=ticket
            )
        except Exception as e:
            logger.error(f"Failed to build analysis prompt: {str(e)}")
            # Fallback to simple prompt
            return f"""
            Analyze this support ticket:
            Subject: {ticket.subject}
            Content: {ticket.content}
            Priority: {ticket.priority}

            Provide classification in JSON format:
            {{
              "category": "technical",
              "sentiment": "negative",
              "urgency": "high",
              "confidence": 0.85,
              "reasoning": "Brief explanation"
            }}
            """

    def _call_ollama(self, prompt: str) -> str:
        """Call Ollama API for ticket analysis.

        Args:
            prompt: The formatted prompt to send to Ollama

        Returns:
            Raw response from Ollama

        Raises:
            requests.RequestException: If API call fails
        """
        backoffs = [2, 4, 8]
        last_err: Exception | None = None
        for attempt, delay in enumerate([0] + backoffs, start=1):
            try:
                response = requests.post(
                    self.ollama_url,
                    json={
                        "model": self.model,
                        "prompt": prompt,
                        "stream": False,
                        "options": {
                            "temperature": 0.1,
                            "top_p": 0.9,
                        },
                    },
                    timeout=60 if attempt == 1 else 30,
                )
                if response.status_code != 200:
                    raise requests.RequestException(
                        f"Ollama API returned status {response.status_code}: {response.text}"
                    )
                result = response.json()
                return result.get("response", "")
            except requests.RequestException as e:
                last_err = e
                logger.error(f"Ollama API call failed (attempt {attempt}): {str(e)}")
                if delay:
                    try:
                        import time as _t
                        _t.sleep(delay)
                    except Exception:
                        pass
        # All attempts failed
        assert last_err is not None
        raise last_err

    def _parse_ollama_response(self, response: str) -> Dict[str, Any]:
        """Parse the JSON response from Ollama.

        Args:
            response: Raw response string from Ollama

        Returns:
            Parsed classification data

        Raises:
            ValueError: If response cannot be parsed as JSON
        """
        try:
            # Try to extract JSON from the response
            # Sometimes LLMs include markdown formatting
            if "```json" in response:
                # Extract JSON from markdown code block
                start = response.find("```json") + 7
                end = response.find("```", start)
                json_str = response[start:end].strip()
            elif "```" in response:
                # Extract JSON from code block
                start = response.find("```") + 3
                end = response.find("```", start)
                json_str = response[start:end].strip()
            else:
                # Try to find JSON in the response
                json_str = response.strip()

            # Parse the JSON
            data = json.loads(json_str)

            # Validate required fields
            required_fields = ["category", "sentiment", "urgency", "confidence"]
            for field in required_fields:
                if field not in data:
                    raise ValueError(f"Missing required field: {field}")

            return data

        except (json.JSONDecodeError, ValueError) as e:
            logger.error(f"Failed to parse Ollama response: {str(e)}")
            logger.error(f"Raw response: {response}")

            # Return default classification on parsing failure
            return {
                "category": "general",
                "sentiment": "neutral",
                "urgency": "medium",
                "confidence": 0.5,
                "reasoning": "Default classification due to parsing error"
            }

    class ClassifyTicketNode(Node):
    """Processes and validates ticket classification results.

    This node takes the raw analysis results from AnalyzeTicketNode
    and converts them into structured classification data. It also
    applies business logic for classification validation and enhancement.
    """

    def process(self, task_context: TaskContext) -> TaskContext:
        """Process the ticket classification workflow.

        Args:
            task_context: The task context containing analysis results

        Returns:
            Updated TaskContext with validated classification
        """
        ticket = task_context.event

        logger.info(f"Starting ticket classification", extra={
            "ticket_id": ticket.id,
            "node": self.node_name,
            "status": "started"
        })

        try:
            # Get analysis results from previous node
            analysis_node = task_context.nodes.get("AnalyzeTicketNode", {})

            if analysis_node.get("status") != "completed":
                raise ValueError("Analysis node did not complete successfully")

            # Extract classification data
            raw_classification = analysis_node.get("classification", {})

            # Validate and enhance classification
            classification_data = self._validate_classification(raw_classification)

            # Apply business logic for classification enhancement
            enhanced_classification = self._enhance_classification(
                classification_data, ticket
            )

            # Store results in task context
            task_context.nodes[self.node_name] = {
                "status": "completed",
                "classification": enhanced_classification,
                "original_analysis": raw_classification
            }

            logger.info(f"Ticket classification completed", extra={
                "ticket_id": ticket.id,
                "node": self.node_name,
                "status": "completed",
                "category": enhanced_classification.category,
                "sentiment": enhanced_classification.sentiment,
                "urgency": enhanced_classification.urgency
            })

        except Exception as e:
            logger.error(f"Ticket classification failed", extra={
                "ticket_id": ticket.id,
                "node": self.node_name,
                "status": "error",
                "error": str(e)
            })

            task_context.nodes[self.node_name] = {
                "status": "error",
                "error": str(e)
            }

        return task_context

    def _validate_classification(self, raw_data: Dict[str, Any]) -> SimpleClassificationSchema:
        """Validate and convert raw classification data to structured format.

        Args:
            raw_data: Raw classification data from AI analysis

        Returns:
            Validated SimpleClassificationSchema object
        """
        # Validate category
        category = raw_data.get("category", "general").lower()
        valid_categories = [
            "technical", "billing", "general", "feature_request", 
            "bug_report", "account", "security"
        ]
        if category not in valid_categories:
            category = "general"

        # Validate sentiment
        sentiment = raw_data.get("sentiment", "neutral").lower()
        valid_sentiments = ["positive", "neutral", "negative"]
        if sentiment not in valid_sentiments:
            sentiment = "neutral"

        # Validate urgency
        urgency = raw_data.get("urgency", "medium").lower()
        valid_urgency_levels = ["low", "medium", "high"]
        if urgency not in valid_urgency_levels:
            urgency = "medium"

        # Validate confidence
        confidence = raw_data.get("confidence", 0.5)
        if not isinstance(confidence, (int, float)) or confidence < 0 or confidence > 1:
            confidence = 0.5

        return SimpleClassificationSchema(
            category=category,
            sentiment=sentiment,
            urgency=urgency,
            confidence=confidence
        )

    def _enhance_classification(
        self, 
        classification: SimpleClassificationSchema, 
        ticket
    ) -> SimpleClassificationSchema:
        """Apply business logic to enhance classification.

        Args:
            classification: Validated classification data
            ticket: Original ticket data

        Returns:
            Enhanced classification with business logic applied
        """
        # Adjust urgency based on ticket priority
        if ticket.priority == "high" and classification.urgency == "low":
            classification.urgency = "medium"
        elif ticket.priority == "low" and classification.urgency == "high":
            classification.urgency = "medium"

        # Adjust sentiment based on urgency
        if classification.urgency == "high" and classification.sentiment == "positive":
            # High urgency with positive sentiment might indicate sarcasm or frustration
            classification.sentiment = "negative"

        # Adjust confidence based on content length
        if len(ticket.content) < 50:
            # Short content might have lower confidence
            classification.confidence = min(classification.confidence, 0.7)

        return classification

    class GenerateResponseNode(Node):
    """Generates helpful responses for customer support tickets.

    This node uses the classification results to generate appropriate,
    helpful responses for customer support tickets. It uses Ollama with
    Llama 3.1 8B for lightweight, local AI response generation.
    """

    def __init__(self):
        self.ollama_url = f"{settings.ollama_base_url}/api/generate"
        self.model = settings.ollama_model
        self.start_time = None

    def process(self, task_context: TaskContext) -> TaskContext:
        """Process the response generation workflow.

        Args:
            task_context: The task context containing ticket and classification data

        Returns:
            Updated TaskContext with generated response
        """
        self.start_time = time.time()
        ticket = task_context.event

        logger.info(f"Starting response generation", extra={
            "ticket_id": ticket.id,
            "node": self.node_name,
            "status": "started"
        })

        try:
            # Get classification from previous node
            classification_node = task_context.nodes.get("ClassifyTicketNode", {})

            if classification_node.get("status") != "completed":
                raise ValueError("Classification node did not complete successfully")

            classification = classification_node.get("classification")

            # Build the response prompt using the template
            prompt = self._build_response_prompt(ticket, classification)

            # Call Ollama API
            response_result = self._call_ollama(prompt)

            # Parse the JSON response
            response_data = self._parse_ollama_response(response_result)

            # Calculate processing time
            processing_time = time.time() - self.start_time

            # Store results in task context
            task_context.nodes[self.node_name] = {
                "status": "completed",
                "response": response_data,
                "processing_time": processing_time,
                "model_used": self.model
            }

            logger.info(f"Response generation completed", extra={
                "ticket_id": ticket.id,
                "node": self.node_name,
                "status": "completed",
                "processing_time": processing_time,
                "response_type": response_data.response_type
            })

        except Exception as e:
            processing_time = time.time() - self.start_time if self.start_time else 0

            logger.error(f"Response generation failed", extra={
                "ticket_id": ticket.id,
                "node": self.node_name,
                "status": "error",
                "error": str(e),
                "processing_time": processing_time
            })

            task_context.nodes[self.node_name] = {
                "status": "error",
                "error": str(e),
                "processing_time": processing_time
            }

        return task_context

    def _build_response_prompt(self, ticket, classification) -> str:
        """Build the response generation prompt using the Jinja2 template.

        Args:
            ticket: The support ticket data
            classification: The classification results

        Returns:
            Formatted prompt string
        """
        try:
            return PromptManager.get_prompt(
                "response_generation",
                ticket=ticket,
                classification=classification
            )
        except Exception as e:
            logger.error(f"Failed to build response prompt: {str(e)}")
            # Fallback to simple prompt
            return f"""
            Generate a helpful response for this support ticket:

            Subject: {ticket.subject}
            Content: {ticket.content}
            Priority: {ticket.priority}
            Category: {classification.category}
            Sentiment: {classification.sentiment}
            Urgency: {classification.urgency}

            Provide response in JSON format:
            {{
              "response_text": "Your helpful response here...",
              "response_type": "automated",
              "suggested_actions": ["action1", "action2"],
              "confidence": 0.9
            }}
            """

    def _call_ollama(self, prompt: str) -> str:
        """Call Ollama API for response generation.

        Args:
            prompt: The formatted prompt to send to Ollama

        Returns:
            Raw response from Ollama

        Raises:
            requests.RequestException: If API call fails
        """
        backoffs = [2, 4, 8]
        last_err: Exception | None = None
        for attempt, delay in enumerate([0] + backoffs, start=1):
            try:
                response = requests.post(
                    self.ollama_url,
                    json={
                        "model": self.model,
                        "prompt": prompt,
                        "stream": False,
                        "options": {
                            "temperature": 0.3,
                            "top_p": 0.9,
                        },
                    },
                    timeout=60 if attempt == 1 else 30,
                )
                if response.status_code != 200:
                    raise requests.RequestException(
                        f"Ollama API returned status {response.status_code}: {response.text}"
                    )
                result = response.json()
                return result.get("response", "")
            except requests.RequestException as e:
                last_err = e
                logger.error(f"Ollama API call failed (attempt {attempt}): {str(e)}")
                if delay:
                    try:
                        import time as _t
                        _t.sleep(delay)
                    except Exception:
                        pass
        assert last_err is not None
        raise last_err

    def _parse_ollama_response(self, response: str) -> ResponseSchema:
        """Parse the JSON response from Ollama.

        Args:
            response: Raw response string from Ollama

        Returns:
            Parsed ResponseSchema object

        Raises:
            ValueError: If response cannot be parsed as JSON
        """
        try:
            # Try to extract JSON from the response
            if "```json" in response:
                # Extract JSON from markdown code block
                start = response.find("```json") + 7
                end = response.find("```", start)
                json_str = response[start:end].strip()
            elif "```" in response:
                # Extract JSON from code block
                start = response.find("```") + 3
                end = response.find("```", start)
                json_str = response[start:end].strip()
            else:
                # Try to find JSON in the response
                json_str = response.strip()

            # Parse the JSON
            data = json.loads(json_str)

            # Validate required fields
            required_fields = ["response_text", "response_type", "confidence"]
            for field in required_fields:
                if field not in data:
                    raise ValueError(f"Missing required field: {field}")

            # Create ResponseSchema object
            return ResponseSchema(
                response_text=data["response_text"],
                response_type=data["response_type"],
                suggested_actions=data.get("suggested_actions", []),
                confidence=data["confidence"]
            )

        except (json.JSONDecodeError, ValueError) as e:
            logger.error(f"Failed to parse Ollama response: {str(e)}")
            logger.error(f"Raw response: {response}")

            # Return default response on parsing failure
            return ResponseSchema(
                response_text="Thank you for contacting support. We have received your request and will get back to you shortly. If this is urgent, please contact our support team directly.",
                response_type="escalation",
                suggested_actions=["Contact support team", "Check status page"],
                confidence=0.5
            )
